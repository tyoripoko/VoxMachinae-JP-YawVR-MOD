# ════════════════════════════════════════════════════════════════
# VoxTelemetry 武器振動設定ファイル (Weapon_Vib.txt)
# ════════════════════════════════════════════════════════════════
# 
# 【YawVR振動モーター（ERM）の物理特性】
#   低HZ（低速回転）→ 指向性のある大きな揺れ（ドカーン！）
#   高HZ（高速回転）→ 指向性が相殺された細かい振動（ブルブル）
#   AMP → 回転の強さ（高すぎると飽和、50前後が効果的）
#   DUR → 揺れの持続時間（秒）
#         ※レーザー（SOUND系）のみ「照射停止後の持続秒数」として扱われる
#           Muzzle音が来るたびにリセット→照射中はずっと一定振動を維持
#
# 【フォーマット】
#   武器名|AMP|HZ|秒|FIRE      → Weapon.Fire() でフック
#   サウンド名|AMP|HZ|秒|SOUND  → VoxSound.Play でフック
#   定数名|値|CONST            → 定数の上書き
#
# 【使い方】
#   1. このファイルを Mods フォルダに配置
#   2. ファイルを保存すると2秒以内に自動リロード
#
# 【GameLink Multiplier設定】
#   Weapon_Vib_Amp = 3.0, Weapon_Vib_Hz = 3.0
#
# ════════════════════════════════════════════════════════════════

# ────────────────────────────────────────────────────────────────

# Railgunチャージ: 高HZ（細かい）→ 低HZ（重い）へ変化
RAIL_CHARGE_TIME|2.7|CONST

RAIL_START_HZ|60|CONST

# Railgunチャージのカーブ指数（Pow適用。t=0から変化し始める）
# 0.5 = 序盤で素早く立ち上がり、後半はなだらか
# 1.0 = 一定ペースで線形増加
# 1.5 = 序盤は緩やか、後半に加速（デフォルト）
# 2.0 = さらに後半集中
# 3.0 = チャージ完了直前まで緩やか、最後に急上昇
RAIL_CURVE|0.4|CONST

# Drill: 連続した中間的な振動
DRILL_AMP|100|CONST
DRILL_HZ|10|CONST

# 未登録武器のデフォルト
DEFAULT_AMP|35|CONST
DEFAULT_HZ|8|CONST
DEFAULT_DUR|0.4|CONST

# レーザー振動のデフォルトタイムアウト（秒）
# 個別エントリの秒フィールドで上書き可能。
# ゲームの最短再発火間隔が約3秒なので 3.5 以上を推奨。
LASER_VIB_TIMEOUT|3.5|CONST
